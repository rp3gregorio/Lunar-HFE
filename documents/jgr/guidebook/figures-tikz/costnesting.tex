% ---- innermost (teal): one Crank--Nicolson hour + the two steps ----
\node[box, draw=teal, fill=teal!12, text width=82mm, inner sep=7pt] (B3) at (0,0) {%

% ---- outer (forest): the K_d sweep ----

  {\sffamily\bfseries 1\,$\cdot$\,Retrieval — sweep $K_d$}\\[2pt]
  {\color{dim}try ${\sim}30$ trial values; keep the one with the lowest RMSE}};

  \node[box, draw=forest, fill=forest!7, fit=(m1)(B2), inner sep=8pt] (B1) {};
\end{scope}
\node[cnt] at ([shift={(-7pt,-6pt)}]B1.north east) {$\times\,30$};

% ---- result ----
\node[font=\sffamily\bfseries\color{char}, anchor=north] (res) at ([yshift=-10pt]B1.south)
