% Shared preamble for standalone-figure compilation.
% Mirrors the colors, libraries, and font setup used by guidebook.tex
% so that a figure compiled on its own looks identical to the one
% embedded in the book.

\usepackage[T1]{fontenc}
\usepackage{newtxtext,newtxmath}
\usepackage[scaled=0.90]{helvet}
\usepackage{amsmath}
\usepackage{xcolor}
\usepackage{tikz}
\usetikzlibrary{positioning, arrows.meta, calc, fit, backgrounds}

% --- book palette (must match guidebook.tex) ---
\definecolor{forest}{HTML}{3D6E4A}
\definecolor{coral}{HTML}{B85B3A}
\definecolor{teal}{HTML}{2C6E73}
\definecolor{char}{HTML}{2B2B2B}
\definecolor{dim}{HTML}{6B6B6B}

% In standalone mode \ref{eq:foo} cannot resolve to a real equation
% number (no .aux from the book). Print the label so the figure layout
% stays correct during iteration.
\providecommand{\bookref}[1]{\texttt{\small[#1]}}
\renewcommand{\ref}[1]{\bookref{#1}}
