% Fig — the geometric grid as finite-volume cells: centers, faces, dz_c,
% and the harmonic-mean face conductivity. Body only.
\begin{tikzpicture}[font=\footnotesize, every node/.style={align=center},
  lbl/.style={font=\scriptsize\color{dim}}]
  % cells widening geometrically (widths 0.55,0.70,0.95,1.35,2.0,3.0)
  \def\ys{0}
  \foreach \w/\i [remember=\xr as \xl (initially 0)] in
      {0.55/0, 0.75/1, 1.05/2, 1.5/3, 2.15/4, 3.1/5} {
    \pgfmathsetmacro\xr{\xl+\w}
    \draw[line width=0.9pt, color=char] (\xl,0) rectangle (\xr,1.0);
    \pgfmathsetmacro\xc{(\xl+\xr)/2}
    \fill[teal] (\xc,0.5) circle (1.6pt);
    \node[lbl, below=0pt] at (\xc,0.42) {$T_{\i}$};
    \global\let\xr\xr
  }
  \node[lbl, anchor=west] at (9.3,0.5)
    {cells grow 8\%\\ per step $\to$ 69 cells\\ reach $z=5$ m};
  % surface + axis
  \draw[line width=1.4pt, color=coral] (0,-0.05) -- (0,1.05);
  \node[font=\scriptsize\color{coral}, anchor=east] at (-0.12,0.5)
    {surface\\ $z=0$};
  \draw[-{Stealth[length=2mm]}, line width=0.8pt, color=dim]
    (0,-0.5) -- (9.2,-0.5) node[below=2pt, pos=0.97, font=\scriptsize\color{dim}] {$z$};
  % dz_c between two centers
  \draw[{Stealth[length=1.6mm]}-{Stealth[length=1.6mm]}, color=forest, line width=0.9pt]
    (0.925,1.25) -- (1.675,1.25);
  \node[lbl, color=forest, above=1pt] at (1.3,1.28)
    {$\Delta z_{c}$: center-to-center};
  % face annotation with harmonic mean
  \draw[color=plumx, line width=1.2pt] (1.3,-0.05) -- (1.3,1.05);
  \node[font=\scriptsize\color{plumx}, anchor=north west, text width=86mm] at (0,-0.85)
    {flux crosses the \textbf{face} between cells --- two conductivities in
     series $\Rightarrow$ harmonic mean
     $K_{\rm face}=2K_0K_1/(K_0+K_1)$};
  \node[lbl, anchor=north west, text width=86mm] at (0,-1.65)
    {fine cells where $T$ curves hard (millimeters at the surface), coarse
     where it is nearly linear (deep)};
\end{tikzpicture}
