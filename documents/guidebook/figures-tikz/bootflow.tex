% Fig — how one bootstrap draw works (and how 1500 of them become a CI).
% Algorithm from retrieve_kd.bootstrap_kd_with_depth_uncertainty (n_boot=1500,
% depth sigma 2.5 cm, seed 42; re-uses the cached K_d-sweep profiles).
% Body only; the standalone wrapper supplies \documentclass + _preamble.
\begin{tikzpicture}[
  font=\sffamily\footnotesize,
  every node/.style={align=center},
  io/.style={rectangle, rounded corners=10pt, draw=dim, line width=0.9pt,
             fill=black!3, inner sep=5pt, text width=7.0cm},
  box/.style={rectangle, rounded corners=2pt, draw=dim, line width=0.8pt,
              fill=white, inner sep=5pt, text width=6.4cm},
  lbox/.style={box, draw=plumc},
  res/.style={rectangle, rounded corners=3pt, draw=forest, fill=forest!12,
              line width=1pt, inner sep=5pt, text width=7.0cm},
  arrow/.style={-{Stealth[length=2.4mm]}, line width=0.9pt, draw=dim,
                rounded corners=3pt}]
\providecolor{plumc}{HTML}{5A4A6A}

  \node[io] (start)
    {\textbf{Inputs.} The $N$ deep sensors ($N{=}7$ A15, $N{=}16$ A17) and the
     29/32 \emph{cached} sweep profiles $\langle T\rangle(z;K_d)$ --- no new solves, ever.};

  \node[box, below=11mm of start] (b1)
    {\textbf{resample}: draw $N$ sensors \emph{with replacement}\\
     {\scriptsize some sensors appear twice, others not at all}};
  \node[box, below=4.5mm of b1] (b2)
    {\textbf{jitter}: nudge each depth by $\mathcal{N}(0,\,2.5\ \mathrm{cm})$\\
     {\scriptsize the Nagihara (2018) emplacement uncertainty}};
  \node[box, below=4.5mm of b2] (b3)
    {\textbf{refit}: RMSE$(K_d)$ of the perturbed dataset from the cached
     profiles $\rightarrow$ parabola vertex};
  \node[lbox, below=4.5mm of b3] (b4)
    {\textbf{record} one plausible $K_d^{*}$ --- draw $k$ of 1500};

  \node[io, below=11mm of b4] (cloud)
    {\textbf{1500 draws = the cloud} (seed 42, archived). Sort it:\\
     median $\rightarrow$ the headline; middle 95\% $\rightarrow$ the CI.};
  \node[res, below=5mm of cloud]
    {medians 4.63 / 7.08;\; CIs $[4.18,6.96]$ / $[6.16,8.07]$\\
     paired A17$-$A15 draws $\rightarrow$ contrast 2.31,\; $p=0.031$};

  \draw[arrow] (start) -- (b1);
  \draw[arrow] (b1) -- (b2);
  \draw[arrow] (b2) -- (b3);
  \draw[arrow] (b3) -- (b4);
  \draw[arrow] (b4) -- (cloud)
    node[midway, right=2pt, font=\sffamily\scriptsize\color{dim}] {after 1500 repeats};
  \draw[arrow] (cloud) -- ($(cloud.south)+(0,-4.5mm)$);

  % loop-back: repeat 1500x (left channel)
  \path (b4.west) -- ++(-1.15,0) coordinate (turn);
  \draw[arrow] (b4.west) --
    node[above=1pt, font=\sffamily\scriptsize\color{dim}] {$\times$1500}
    (turn) |- (b1.west);

  % container around the per-draw loop
  \begin{scope}[on background layer]
    \node[fill=black!4, rounded corners=3pt, fit=(b1)(b4), inner sep=3mm] (cont) {};
  \end{scope}
  \node[anchor=south west, font=\sffamily\small\bfseries\color{plumc}]
    at ([yshift=1.5mm]cont.north west) {ONE BOOTSTRAP DRAW \textbullet\ cheap: no solver call};

\end{tikzpicture}
