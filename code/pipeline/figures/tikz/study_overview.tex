% --- PROCESS ---
\node[prb] at (6.3,-0.4) (pA) {%
  \textbf{A.\; Stability windows\; $\to T_{\rm eq}$}\\[1pt]
  one window per sensor; sensors below 80 cm:\\
  {\color{dim}7 at A15 (84--139 cm), 16 at A17 (130--234 cm)}};
\node[prb, below=3mm of pA.south west, anchor=north west] (pB) {%
  \textbf{B.\; Forward model}\; $\to T(z),\ \langle T_s\rangle,\ \kappa$\\[1pt]
  heat equation $\rho\,c_p\,\partial_t T=\partial_z(K\,\partial_z T)$\\
  solved to its periodic steady state\\
  (flux-anchored solver), for each $A$ and $K_d$};
\node[prb, below=3mm of pB.south west, anchor=north west] (pC) {%
  \textbf{C.\; Each probe and each thermometer type}\\[1pt]
  the fit of D on each probe (its own diffusivity)\\
  and on each thermometer type (TG, TR)\\
  $K_d=4.5$--6.2, each rejecting 3.4\\
  {\color{dim}A15 probe 2 (two sensors): albedo held}};
\node[prb, below=3mm of pC.south west, anchor=north west, draw=forest, line width=0.9pt] (pD) {%
  \textbf{D.\; All sensors together: fit $A$ and $K_d$}\\[1pt]
  minimize\; $J=$ sensors $+$ surface mean $+$ diffusivity\\
  on 25 albedos $\times$ 94 values of $K_d$\\[2pt]
  {\color{forest}\itshape The temperatures alone trade $A$ against $K_d$;\\
  $\kappa=K/(\rho\,c_p)$ does not depend on $A$ or $Q_b$,\\
  so it fixes $K_d$.}};
\node[prb, below=3mm of pD.south west, anchor=north west] (pE) {%
  \textbf{E.\; One measurement at a time, and checks}\\[1pt]
  albedo fixed: $K_d$ follows the albedo; albedo free:\\
  a long valley; annual wave alone: 5.7, 5.6\\
  likelihood-ratio tests, bootstrap, error budget, MCMC\\
  {\color{dim}record choices, own diffusivities, warming, Diviner}};

\node[rsb, below=2mm of r3.south west, anchor=north west] (r4) {%
